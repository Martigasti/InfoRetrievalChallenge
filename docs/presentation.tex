% SLIDE: Component technical details
\begin{frame}{1. Pipeline Overview: What Each Component Does}
\begin{itemize}
    \item \textbf{SPECTER2 + proximity adapter} (citation trained)
    \begin{itemize}
        \item \small Input: \texttt{title [SEP] abstract + body chunks}
        \item \small CLS pooling, L2 normalised, 768 dims
    \end{itemize}
    \vspace{0.3em}
    \item \textbf{BGE large v1.5} (general semantic similarity)
    \begin{itemize}
        \item \small Input: \texttt{title + abstract + body chunks}
        \item \small Mean pooling, L2 normalised, 1024 dims
    \end{itemize}
    \vspace{0.3em}
    \item \textbf{BM25Okapi} (lexical, exact matches)
    \begin{itemize}
        \item \small Porter stemmed, stopword filtered
        \item \small Indexes the full enriched text (no 512 token cap)
    \end{itemize}
    \vspace{0.3em}
    \item \textbf{Weighted RRF}: $\text{score}(d) = \sum_r w_r / (k + \text{rank}_r(d))$, $k=10$
\end{itemize}
\end{frame}

% SLIDE: Signal orthogonality
\begin{frame}{1. Pipeline Overview: Why Three Different Retrievers?}
The three components capture \textbf{orthogonal signals}. A paper can be a
valid citation for entirely different reasons.

\vspace{0.6em}
\centering
\small
\begin{tabular}{lll}
\toprule
\textbf{Retriever} & \textbf{Question it answers} & \textbf{Catches} \\
\midrule
SPECTER2 & ``Would a scientist cite this?''  & citation graph neighbours \\
BGE large & ``Is this semantically similar?'' & paraphrase, related topics \\
BM25 & ``Do they share exact terms?''         & tool / dataset / author names \\
\bottomrule
\end{tabular}

\vspace{0.8em}
\raggedright
\small
\textbf{Concrete example.}
A biology paper cites ``PyTorch''.
\begin{itemize}
    \item BM25: \textcolor{improvement}{matches} on the string ``PyTorch''
    \item SPECTER2 / BGE: \textcolor{failure}{miss} (different domain in embedding space)
\end{itemize}
Another paper cites a closely related method paper using different vocabulary.
\begin{itemize}
    \item BGE: \textcolor{improvement}{matches} on semantic similarity
    \item BM25: \textcolor{failure}{miss} (no lexical overlap)
\end{itemize}

Fusion is only valuable when retrievers \emph{disagree}.
\end{frame}

\begin{frame}{\textbf{2. Ablation Study: Contribution of Each Component}}
\centering
\small
\begin{tabular}{lcc}
\toprule
\textbf{Configuration} & \textbf{NDCG@10} & \textbf{$\Delta$} \\
\midrule
TF-IDF baseline                          & 0.484 & \textit{baseline}  \\
MiniLM dense baseline                    & 0.500 & \textcolor{improvement}{$+0.016$} \\
\quad Swap to SPECTER2 + proximity       & 0.540 & \textcolor{improvement}{$+0.040$} \\
\quad Add BM25 (Weighted RRF)            & 0.568 & \textcolor{improvement}{$+0.028$} \\
\quad Add body chunks                    & 0.577 & \textcolor{improvement}{$+0.009$} \\
\quad Add BGE large as 2nd dense         & 0.585 & \textcolor{improvement}{$+0.008$} \\
\quad Wider pool (200 $\to$ 300)         & 0.587 & \textcolor{improvement}{$+0.002$} \\
\bottomrule
\end{tabular}

\vspace{0.8em}
\begin{itemize}
    \item \textbf{Largest gain}: citation aware encoder (SPECTER2), $+0.040$
    \item \textbf{Second largest}: lexical fusion (BM25), $+0.028$
    \item Diminishing returns after 3 way fusion
\end{itemize}
\end{frame}

% SLIDE: Tuning details
\begin{frame}{2. Ablation Study: Tuning Decisions}
\textbf{BM25 weight grid search.} Tried $w_{\text{bm25}} \in \{0.3, 0.5, 0.7, 1.0\}$
with $w_{\text{specter}} = w_{\text{bge}} = 1.0$ fixed.

\vspace{0.3em}
\centering
\small
\begin{tabular}{lcc}
\toprule
\textbf{Chunks per doc} & \textbf{Optimal $w_{\text{bm25}}$} & \textbf{NDCG@10} \\
\midrule
3 chunks  & 0.5 & 0.585 \\
6 chunks  & 1.0 & 0.587 \\
\bottomrule
\end{tabular}

\vspace{0.6em}
\raggedright
\small
With more body text indexed, BM25 becomes as informative as the dense
retrievers and earns an equal weight.

\vspace{0.8em}
\textbf{RRF constant $k$.} Swept $\{10, 60, 100\}$. $k=10$ preserves top rank
contributions best for a small candidate pool; higher $k$ flattens rank
differences.

\vspace{0.6em}
\textbf{Why no learned weights?} 100 public queries $\Rightarrow$ fitting
weights overfits to the validation set. Weighted RRF with a handful of
discrete choices is more robust.
\end{frame}

% PART 4: Error Analysis

\begin{frame}{\textbf{4. Error Analysis: What Did Not Work}}
\begin{itemize}
    \item \textbf{\textcolor{failure}{Cross-encoder reranking}}\\
    \footnotesize bge-reranker-v2-m3, SciBERT, MS-MARCO rerankers on the top 20.
    \vspace{0.3em}

    \item \textbf{\textcolor{failure}{HyDE}}\\
    \footnotesize flan-t5-base hypothetical abstracts embedded by SPECTER2 / BGE.
    \vspace{0.3em}

    \item \textbf{\textcolor{failure}{BM25 pseudo-relevance feedback}}\\
    \footnotesize RM3-style query expansion from top fused documents.
    \vspace{0.3em}

    \item \textbf{\textcolor{failure}{Jina embeddings v3}} \hfill $\Delta = -0.012$\\
    \footnotesize Top of MTEB, but trained for web / multilingual retrieval.
    \vspace{0.3em}

    \item \textbf{\textcolor{failure}{SciNCL as 4th retriever}} \hfill $\Delta = -0.004$\\
    \footnotesize Redundant with SPECTER2 — both trained on citation graphs.
\end{itemize}
\end{frame}

% ===============================================================
% Surprises we encountered
% ===============================================================
\begin{frame}{\textbf{Surprises We Encountered}}
\begin{itemize}
    \item \textbf{MTEB ranking did not predict task performance.}\\
    \footnotesize Jina v3 sits near the top of MTEB yet lost $-0.012$ NDCG@10
    against BGE. Leaderboard ranking is a benchmark artefact, not a guarantee
    on an unseen task.
    \vspace{0.4em}

    \item \textbf{A small, task-aligned model beat larger general models.}\\
    \footnotesize SPECTER2 base (110M params) outperformed every general
    encoder we tried. Training objective $>$ parameter count.
    \vspace{0.4em}

    \item \textbf{BM25 weight shifted with body chunk count.}\\
    \footnotesize Optimal $w_{\text{bm25}} = 0.5$ at 3 chunks, $1.0$ at 6 chunks.
    More indexed body text makes BM25 as informative as the dense retrievers
    — an interaction we did not expect before grid searching.
    \vspace{0.4em}

    \item \textbf{Two citation-trained encoders were near-redundant.}\\
    \footnotesize SciNCL and SPECTER2 agreed too much for RRF to exploit
    their differences. We expected ``more citation signal = better''; we got
    a null result.
    \vspace{0.4em}

    \item \textbf{Simple RRF beat learned fusion.}\\
    \footnotesize LightGBM LambdaRank had the right objective (NDCG@10) and
    richer features, but 100 queries was not enough — CV overfit, public
    score collapsed.
\end{itemize}
\end{frame}

% ===============================================================
% PART 5: What We Learned
% ===============================================================
\begin{frame}{\textbf{5. What We Learned}}
\begin{enumerate}
    \item \textbf{Task aligned models matter most.}\\
    \footnotesize Switching to SPECTER2 (citation trained) gave $+0.040$, the
    largest single jump. Generic MTEB winners underperformed.

    \vspace{0.5em}
    \item \textbf{Fusion gains come from disagreement, not from more models.}\\
    \footnotesize SPECTER2 + BGE + BM25 captures three different signals:
    citation proximity, semantic similarity, lexical overlap. Adding a 4th
    citation encoder (SciNCL) changed nothing.

    \vspace{0.5em}
    \item \textbf{More surface text helps, up to a point.}\\
    \footnotesize Body chunks helped BM25 pick up method / dataset names
    missing from abstracts. Returns flattened beyond 6 chunks.

    \vspace{0.5em}
    \item \textbf{Weighted RRF is simple and effective.}\\
    \footnotesize One hyperparameter per retriever, no learned weights to
    overfit, no score normalisation needed across heterogeneous retrievers.
\end{enumerate}
\end{frame}
